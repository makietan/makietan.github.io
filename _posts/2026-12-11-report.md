mpv を使っていて、プレイリストを操作したい欲求が抑えられず、作成したスクリプト。

## save_playlist.lua

```lua
-- save_playlist.lua
local mp = require 'mp'

local title_cache = {}
local fetching_urls = {}

-- プレイリスト連携モード
local is_playlist_mode = false

local home = os.getenv("HOME")
local save_path = home .. "/.config/mpv/current_playlist.m3u"

function save_playlist_core()
    if not is_playlist_mode then return end

    local playlist = mp.get_property_native("playlist")
    if not playlist or #playlist == 0 then return end

    local file, err = io.open(save_path, "w")
    if not file then return end

    file:write("#EXTM3U\n")

    for _, item in ipairs(playlist) do
        if item.filename then
            local title = item.title or title_cache[item.filename]
            if title and title ~= "" then
                file:write("#EXTINF:-1," .. title .. "\n")
            end
            file:write(item.filename .. "\n")
        end
    end

    file:close()
end

-- ==========================================
-- [手動] F5キー (保存)
-- ==========================================
mp.register_script_message("save-playlist-custom", function()
    is_playlist_mode = true
    save_playlist_core()
    mp.osd_message("プレイリストを保存しました (F5)")
end)

-- ==========================================
-- [手動] F7キー (シームレス読み込み)
-- ==========================================
mp.register_script_message("load-playlist-custom", function()
    is_playlist_mode = true
    local current_path = mp.get_property("path")
    
    -- 動画が何も再生されていない場合は、そのまま通常の読み込み
    if not current_path then
        mp.commandv("loadlist", save_path, "replace")
        mp.osd_message("プレイリストを読み込みました (F7)")
        return
    end

    -- 1. 現在再生中のファイル「以外」をリストから消去（※再生は絶対に止まらない）
    mp.command_native({"playlist-clear"})

    -- 2. 編集済みのM3Uをリストの末尾にこっそり追加
    mp.command_native({"loadlist", save_path, "append"})
    
    -- 3. 重複した現在再生中のファイルを削除し、VS Codeで並び替えた正しい位置に移動させる
    local playlist = mp.get_property_native("playlist")
    local duplicate_index = -1
    
    if playlist and #playlist > 1 then
        for i = 2, #playlist do
            -- プレイリスト内のパスと現在再生中のパスが一致するか確認
            if playlist[i].filename == current_path then
                duplicate_index = i - 1 -- mpvのコマンド用インデックス(0始まり)に変換
                break
            end
        end

        if duplicate_index ~= -1 then
            -- 読み込んだリスト側の重複分を消す
            mp.commandv("playlist-remove", tostring(duplicate_index))
            
            -- 現在再生中のアイテム(index:0)を、本来あるべき正しい順番に移動する
            local target_index = duplicate_index - 1
            if target_index > 0 then
                mp.commandv("playlist-move", "0", tostring(target_index))
            end
        end
    end

    mp.osd_message("リストをシームレスに更新しました (F7)")
end)

-- ==========================================
-- URL自動取得 (勝手にファイル保存はしない)
-- ==========================================
mp.observe_property("playlist", "native", function(name, playlist)
    if not playlist then return end
    
    for _, item in ipairs(playlist) do
        local url = item.filename
        if url and url:match("^https?://") and not item.title and not title_cache[url] and not fetching_urls[url] then
            fetching_urls[url] = true
            
            mp.command_native_async({
                name = "subprocess",
                capture_stdout = true,
                playback_only = false,
                args = {"yt-dlp", "--no-warnings", "--get-title", url}
            }, function(success, result, error)
                if success and result.status == 0 and result.stdout ~= "" then
                    local fetched_title = result.stdout:gsub("[\r\n]", "")
                    -- タイトルをメモリに保持。ファイルへの書き出しはF5を押すまで待機
                    title_cache[url] = fetched_title
                end
            end)
        end
    end
end)

-- ==========================================
-- 自動保存 (リスト完走時 ＆ 終了時)
-- ==========================================
mp.register_event("end-file", function(e)
    if e.reason == "eof" and is_playlist_mode then
        local pos = mp.get_property_number("playlist-pos")
        local count = mp.get_property_number("playlist-count")
        if pos ~= nil and count ~= nil and pos == (count - 1) then
            save_playlist_core()
        end
    end
end)

mp.register_event("shutdown", save_playlist_core)
```

## input.conf

```conf
# ==========================================
# 1. 誤終了防止
# ==========================================
q ignore
Q ignore

# ==========================================
# 2. プレイリストの操作設定 (F5 〜 F8)
# ==========================================
# 【保存】(F5) 現在のプレイリストをファイルに書き出す
F5 script-message save-playlist-custom

# 【編集】(F6) VS Code でプレイリストを開く
F6 run "/bin/sh" "-c" "code ~/.config/mpv/current_playlist.m3u"

# 【リロード】(F7) スクリプト経由で読み込み、連携モードをONにする
F7 script-message load-playlist-custom

# 【クリア】(F8) 現在のプレイリストを空にする（※誤操作防止のため端に配置）
F8 playlist-clear

# ==========================================
# 3. フォルダを開く (F9)
# ==========================================
# 【設定フォルダを開く】(F9) ~/.config/mpv を VS Code で開く
F9 run "/bin/sh" "-c" "code ~/.config/mpv/"
```
