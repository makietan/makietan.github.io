<script>
  // デフォルト画像設定（初期表示用）
  const defaultBefore = "https://picsum.photos/id/10/600/400";
  const defaultAfter = "https://picsum.photos/id/20/600/400";

  // 画面上の全画像要素の更新関数
  function updateAllImages(beforeSrc, afterSrc) {
    if (beforeSrc) {
      document.getElementById('swipeBeforeImg').src = beforeSrc;
      document.getElementById('onionBeforeImg').src = beforeSrc;
      document.getElementById('diffBeforeImg').src = beforeSrc;
    }
    if (afterSrc) {
      document.getElementById('swipeAfterImg').src = afterSrc;
      document.getElementById('onionAfterImg').src = afterSrc;
      document.getElementById('diffAfterImg').src = afterSrc;
    }
  }

  // 初期画像の読み込み
  updateAllImages(defaultBefore, defaultAfter);

  // --- ドラッグ＆ドロップ/ファイル選択の制御 ---
  function setupUploader(zoneId, inputId, thumbId, isBefore) {
    const zone = document.getElementById(zoneId);
    const input = document.getElementById(inputId);
    const thumb = document.getElementById(thumbId);

    // クリックでファイルダイアログを開く
    zone.addEventListener('click', () => input.click());

    // ドラッグ時の表示変化
    ['dragenter', 'dragover'].forEach(eventName => {
      zone.addEventListener(eventName, (e) => {
        e.preventDefault();
        e.stopPropagation();
        zone.classList.add('dragover');
      });
    });

    ['dragleave', 'drop'].forEach(eventName => {
      zone.addEventListener(eventName, (e) => {
        e.preventDefault();
        e.stopPropagation();
        zone.classList.remove('dragover');
      });
    });

    // ドロップ処理
    zone.addEventListener('drop', (e) => {
      const files = e.dataTransfer.files;
      if (files.length > 0) handleFile(files[0]);
    });

    // ファイル選択処理
    input.addEventListener('change', (e) => {
      if (e.target.files.length > 0) handleFile(e.target.files[0]);
    });

    // ファイル読み込み処理
    function handleFile(file) {
      if (!file.type.startsWith('image/')) return;
      const reader = new FileReader();
      reader.onload = (e) => {
        const src = e.target.result;
        thumb.src = src;
        thumb.hidden = false;
        if (isBefore) {
          updateAllImages(src, null);
        } else {
          updateAllImages(null, src);
        }
      };
      reader.readAsDataURL(file);
    }
  }

  setupUploader('beforeZone', 'beforeInput', 'beforeThumb', true);
  setupUploader('afterZone', 'afterInput', 'afterThumb', false);

  // --- 1. Swipe Diff インタラクション ---
  const swipeContainer = document.getElementById('swipeContainer');
  const swipeWrapper = document.getElementById('swipeWrapper');
  const swipeBar = document.getElementById('swipeBar');
  const swipeAfterImg = document.getElementById('swipeAfterImg');

  function handleSwipe(e) {
    const rect = swipeContainer.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const ratio = Math.max(0, Math.min(1, x / rect.width));
    const percentage = ratio * 100;

    swipeWrapper.style.width = `${percentage}%`;
    swipeBar.style.left = `${percentage}%`;
    // 親要素のレスポンシブ幅に合わせて調整
    swipeAfterImg.style.width = `${rect.width}px`;
  }

  swipeContainer.addEventListener('pointermove', handleSwipe);

  // --- 2. Onion Skin Diff インタラクション ---
  const onionContainer = document.getElementById('onionContainer');
  const onionAfter = document.getElementById('onionAfterImg');

  onionContainer.addEventListener('pointermove', (e) => {
    const rect = onionContainer.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const opacity = Math.max(0, Math.min(1, x / rect.width));

    onionAfter.style.opacity = opacity;
  });
</script>
