<script type="text/javascript">
const STORAGE_KEY = '2026-09-11-report';
const checkboxes = document.querySelectorAll('input[type="checkbox"]');
// チェック状態を localStorage に保存
checkboxes.forEach(checkbox => {
  checkbox.disabled = false;
  checkbox.addEventListener('change', () => {
    const state = {};
    checkboxes.forEach(cb => {
      state[cb.id] = cb.checked;
    });
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  });
});
// ページ読み込み時にチェック状態を復元
const savedState = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
checkboxes.forEach(checkbox => {
  if (savedState[checkbox.id] !== undefined) {
    checkbox.checked = savedState[checkbox.id];
  }
});
// チェックボックスにデフォルトのインデックスを data-default-index 属性として付与する
const checkboxesArray = Array.from(checkboxes);
checkboxesArray.forEach((checkbox, index) => {
  checkbox.dataset.defaultIndex = index;
});
// ソートボタンが押されたときの処理を書く
document.querySelectorAll('.sort-buttons button').forEach(button => {
  button.addEventListener('click', () => {
    const sortType = button.textContent;
    if (sortType === '掲載順') {
      checkboxesArray.sort((a, b) => a.dataset.defaultIndex - b.dataset.defaultIndex);
    } else if (sortType === '名前順') {
      checkboxesArray.sort((a, b) => {
        const nameA = a.parentNode.textContent.replace(/★\d+ /, '');
        const nameB = b.parentNode.textContent.replace(/★\d+ /, '');
        return nameA.localeCompare(nameB);
      });
    } else if (sortType === '衣装順') {
      checkboxesArray.sort((a, b) => {
        const costumeA = a.parentNode.textContent.match(/\[(.*?)\]$/)?.[1] || '';
        const costumeB = b.parentNode.textContent.match(/\[(.*?)\]$/)?.[1] || '';
        return costumeA.localeCompare(costumeB);
      });
    }
    const ul = checkboxesArray[0].closest('ul') || checkboxesArray[0].parentNode;
    checkboxesArray.forEach(checkbox => ul.appendChild(checkbox.parentNode));
  });
});
</script>
